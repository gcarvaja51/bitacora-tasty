// Trend Magic
period_TM = input.int(20, "CCI period")
coeff_TM = input.float(1, "ATR Multiplier")
AP_TM = input.int(5, "ATR Period")
ATR_TM = ta.sma(ta.tr, AP_TM)
src_TM = input.source(close, "TM Source")
upT_TM = low - ATR_TM * coeff_TM
downT_TM = high + ATR_TM * coeff_TM
var float MagicTrend = na
MagicTrend := ta.cci(src_TM, period_TM) >= 0 ? (upT_TM < nz(MagicTrend[1]) ? nz(MagicTrend[1]) : upT_TM) : (downT_TM > nz(MagicTrend[1]) ? nz(MagicTrend[1]) : downT_TM)
color_TM = ta.cci(src_TM, period_TM) >= 0 ? color.new(color.blue, 0) : color.new(color.red, 0)
plot(MagicTrend, color=color_TM, linewidth=3)
alertcondition(ta.cross(close, MagicTrend), title="Cross Alert", message="Price - MagicTrend Crossing!")
alertcondition(ta.crossover(low, MagicTrend), title="CrossOver Alarm", message="BUY SIGNAL!")
alertcondition(ta.crossunder(high, MagicTrend), title="CrossUnder Alarm", message="SELL SIGNAL!")

// CM SlingShot System
emaSlow_SS = ta.ema(close, 20)
emaFast_SS = ta.ema(close, 10)
sae_SS = input.bool(true, title="Show Aggressive Entry?")
sce_SS = input.bool(true, title="Show Conservative Entry?")
st_SS = input.bool(true, title="Show Trend Arrows?")

pullbackUpT_SS = emaFast_SS > emaSlow_SS and close < emaFast_SS
pullbackDnT_SS = emaFast_SS < emaSlow_SS and close > emaFast_SS
entryUpT_SS = emaFast_SS > emaSlow_SS and close[1] < emaFast_SS and close > emaFast_SS
entryDnT_SS = emaFast_SS < emaSlow_SS and close[1] > emaFast_SS and close < emaFast_SS
upTrend_SS = emaFast_SS >= emaSlow_SS
downTrend_SS = emaFast_SS < emaSlow_SS

// Calcular pendiente manualmente
slope_SS = emaFast_SS - emaFast_SS[1]

// MACD
fastLength_MACD = input.int(12, title="MACD Fast Length")
slowLength_MACD = input.int(26, title="MACD Slow Length")
signalSmoothing_MACD = input.int(9, title="MACD Signal Smoothing")

[macdLine, signalLine, _] = ta.macd(close, fastLength_MACD, slowLength_MACD, signalSmoothing_MACD)
macdSlope = macdLine - macdLine[1] // Pendiente del MACD

// Condiciones adicionales para largos y cortos con pendiente del MACD y comparación entre línea rápida y lenta
longCondition = slope_SS > 0 and close > MagicTrend and entryUpT_SS and macdSlope > 0 and macdLine > signalLine
shortCondition = slope_SS < 0 and close < MagicTrend and entryDnT_SS and macdSlope < 0 and macdLine < signalLine

// Mostrar solo las señales que estén alineadas con el MACD
col_SS = emaFast_SS > emaSlow_SS ? color.new(color.lime, 40) : color.new(color.red, 40)
p1_SS = plot(emaSlow_SS, title="Slow MA", style=plot.style_linebr, linewidth=4, color=col_SS)
p2_SS = plot(emaFast_SS, title="Fast MA", style=plot.style_linebr, linewidth=2, color=col_SS)
fill(p1_SS, p2_SS, color=color.silver, transp=50)

barcolor(sae_SS and pullbackUpT_SS and macdSlope > 0 and macdLine > signalLine ? color.yellow : sae_SS and pullbackDnT_SS and macdSlope < 0 and macdLine < signalLine ? color.yellow : na)
barcolor(sce_SS and longCondition ? color.aqua : sce_SS and shortCondition ? color.aqua : na)
plotshape(st_SS and upTrend_SS and macdSlope > 0 and macdLine > signalLine ? upTrend_SS : na, title="Conservative Buy Entry Triangle", style=shape.triangleup, location=location.belowbar, color=color.lime, transp=0)
plotshape(st_SS and downTrend_SS and macdSlope < 0 and macdLine < signalLine ? downTrend_SS : na, title="Conservative Short Entry Triangle", style=shape.triangledown, location=location.abovebar, color=color.red, transp=0)
plotarrow(longCondition ? 1 : na, title="Up Entry Arrow", colorup=color.lime, maxheight=30, minheight=30, transp=0)
plotarrow(shortCondition ? -1 : na, title="Down Entry Arrow", colordown=color.red, maxheight=30, minheight=30, transp=0)

// Chandelier Exit
length_CE = input.int(22, title="ATR Period")
mult_CE = input.float(3.0, title="ATR Multiplier")
showLabels_CE = input.bool(true, title="Show Buy/Sell Labels?")
useClose_CE = input.bool(true, title="Use Close Price for Extremums?")
highlightState_CE = input.bool(true, title="Highlight State?")

atr_CE = mult_CE * ta.atr(length_CE)

longStop_CE = (useClose_CE ? ta.highest(close, length_CE) : ta.highest(length_CE)) - atr_CE
longStopPrev_CE = nz(longStop_CE[1], longStop_CE)
longStop_CE := close[1] > longStopPrev_CE ? math.max(longStop_CE, longStopPrev_CE) : longStop_CE

shortStop_CE = (useClose_CE ? ta.lowest(close, length_CE) : ta.lowest(length_CE)) + atr_CE
shortStopPrev_CE = nz(shortStop_CE[1], shortStop_CE)
shortStop_CE := close[1] < shortStopPrev_CE ? math.min(shortStop_CE, shortStopPrev_CE) : shortStop_CE

var int dir_CE = 1
dir_CE := close > shortStopPrev_CE ? 1 : close < longStopPrev_CE ? -1 : dir_CE

longStopPlot_CE = plot(dir_CE == 1 ? longStop_CE : na, title="Long Stop", style=plot.style_linebr, linewidth=2, color=color.green)
buySignal_CE = dir_CE == 1 and dir_CE[1] == -1
plotshape(buySignal_CE ? longStop_CE : na, title="Long Stop Start", location=location.absolute, style=shape.circle, size=size.tiny, color=color.green, transp=0)
plotshape(buySignal_CE and showLabels_CE ? longStop_CE : na, title="Buy Label", text="Buy", location=location.absolute, style=shape.labelup, size=size.tiny, color=color.green, textcolor=color.white, transp=0)

shortStopPlot_CE = plot(dir_CE == 1 ? na : shortStop_CE, title="Short Stop", style=plot.style_linebr, linewidth=2, color=color.red)
sellSignal_CE = dir_CE == -1 and dir_CE[1] == 1
plotshape(sellSignal_CE ? shortStop_CE : na, title="Short Stop Start", location=location.absolute, style=shape.circle, size=size.tiny, color=color.red, transp=0)
plotshape(sellSignal_CE and showLabels_CE ? shortStop_CE : na, title="Sell Label", text="Sell", location=location.absolute, style=shape.labeldown, size=size.tiny, color=color.red, textcolor=color.white, transp=0)

midPricePlot_CE = plot(ohlc4, title="", style=plot.style_circles, linewidth=0, display=display.none, editable=false)

longFillColor_CE = highlightState_CE ? (dir_CE == 1 ? color.new(color.green, 85) : na) : na
shortFillColor_CE = highlightState_CE ? (dir_CE == -1 ? color.new(color.red, 85) : na) : na
fill(midPricePlot_CE, longStopPlot_CE, title="Long State Filling", color=longFillColor_CE)
fill(midPricePlot_CE, shortStopPlot_CE, title="Short State Filling", color=shortFillColor_CE)

changeCond_CE = dir_CE != dir_CE[1]
alertcondition(changeCond_CE, title="Alert: CE Direction Change", message="Chandelier Exit has changed direction!")
alertcondition(buySignal_CE, title="Alert: CE Buy", message="Chandelier Exit Buy!")
alertcondition(sellSignal_CE, title="Alert: CE Sell", message="Chandelier Exit Sell!")

ema_slow15_B = input.int(20, "EMA Lenta 15m",  group=grpB1)

e20_15_B = ta.ema(c15_B, ema_slow15_B)
e20_15_B_prev = e20_15_B[1]

fase2_15m_B = c15_B > e20_15_B and e10_15_B > e20_15_B and e20_15_B > e20_15_B_prev
fase4_15m_B = c15_B < e20_15_B and e10_15_B < e20_15_B and e20_15_B < e20_15_B_prev

// Alineacion base (sin el patron de retroceso): tendencia + fuerza + marco maestro 15m.
// Usa el ESTADO del MACD (linea vs señal) y no su pendiente bar-a-bar, que oscila de
// signo constantemente incluso dentro de una tendencia sostenida y volvia a "formar"
// la alineacion decenas de veces al dia (ruido, no señal real).
coreAlign_bull_B = close > MagicTrend and emaFast_SS > emaSlow_SS and macdLine > signalLine and fase2_15m_B
coreAlign_bear_B = close < MagicTrend and emaFast_SS < emaSlow_SS and macdLine < signalLine and fase4_15m_B

// Camino B: mientras la alineacion este activa (no exige que el precio haya
// retrocedido bajo la EMA10, a diferencia del retroceso clasico de SlingShot),
// respetando un tiempo minimo entre señales para no repetir en cada barra.
// Ajuste 2026-07-22 (a pedido del usuario): bajado de 20 a 10 min — la ventaja real
// es que hasOpenPosition ya impide 2 trades simultaneos, asi que el gap solo necesita
// evitar spam entre barras, no suplir la exclusividad de posicion.
gapMinutes_B = input.int(10, "Camino B — minutos minimos entre señales", group=grpB1)
gapBars_B = math.round(gapMinutes_B / (timeframe.multiplier))
var int lastFireBull_B = -999
var int lastFireBear_B = -999
enoughGapBull_B = (bar_index - lastFireBull_B) >= gapBars_B
enoughGapBear_B = (bar_index - lastFireBear_B) >= gapBars_B
freshAlign_bull_B = coreAlign_bull_B and enoughGapBull_B
freshAlign_bear_B = coreAlign_bear_B and enoughGapBear_B
if freshAlign_bull_B
    lastFireBull_B := bar_index
if freshAlign_bear_B
    lastFireBear_B := bar_index

// Senal final: solo Camino B (confluencia recien formada) — Camino A (retroceso
// clasico) se desactivo tras el backtest de 58 dias: 48.8% WR y -$130 P&L en 41

sig_bull_B = freshAlign_bull_B

fmtGDV(v) =>
    av = math.abs(v)
    av >= 1000000000 ? str.tostring(v / 1000000000, "#.##") + "B" : av >= 1000000 ? str.tostring(v / 1000000, "#.##") + "M" : str.tostring(v, "#.##")

arrowGDV(now, prev) =>
    now > prev ? " ▲" : now < prev ? " ▼" : " ─"
